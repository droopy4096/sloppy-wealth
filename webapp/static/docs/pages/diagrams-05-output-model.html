    direction TB

    %% ---------- Root: one result per ScenarioSet run (ADR-0028) ----------
    class SetResult {
        +schemaVersion
        +horizon : Horizon
        +scenarios : ScenarioResult[1..*]
        +setGoals : GoalResult[*]
    }
    SetResult "1" *-- "1..*" ScenarioResult
    SetResult "1" *-- "*" GoalResult : set-level

    class ScenarioResult {
        +scenarioId
        +name
        +baseDate : Period (ADR-0013)
        +inflation : RateGenerator (shown, ADR-0028)
        +incomeMode : MODELLED | EXPENSES_COVERED (shown, ADR-0044)
        +appliedDefaults : AppliedDefault[*] (ADR-0040)
    }
    ScenarioResult "1" *-- "1..*" PeriodRow : one per month
    ScenarioResult "1" *-- "*" HoldingSeries : per-holding detail
    ScenarioResult "1" *-- "*" Movement : ledger
    ScenarioResult "1" *-- "*" TallySeries
    ScenarioResult "1" *-- "*" GoalResult : scenario-level
    ScenarioResult "1" *-- "*" Shortfall

    %% ---------- Main table (ADR-0050) ----------
    class PeriodRow {
        +period : Period
        +cash : Money (operating account)
        +investments : Money (other accounts + pension value)
        +propertyAndVehicles : Money (assets)
        +debts : Money (liabilities)
        +assetsNetWorth : Money (investments + property - debts)
        +netWorth : Money (cash + assetsNetWorth)
        +liquidationNetWorth : Money (ADR-0031, excludes pension)
        +real(metric) Money (in baseDate dollars, ADR-0013)
    }

    %% ---------- Detail ----------
    class HoldingSeries {
        +holdingId
        +name
        +kind : ACCOUNT | ASSET | LIABILITY | PENSION
        +group : CASH | INVESTMENTS | PROPERTY | DEBTS
        +points : HoldingPoint[1..*]
    }
    HoldingSeries "1" *-- "1..*" HoldingPoint
    class HoldingPoint {
        +period : Period
        +opening : Money
        +closing : Money
        +costBasis : Money [0..1] (assets)
        +interest : Money [0..1] (loans)
        +principalPaid : Money [0..1] (loans)
        +rate : Decimal [0..1] (current annual rate)
    }

    class Movement {
        +period : Period
        +kind : MovementKind
        +from : HoldingRef [0..1] (none = outside)
        +to : HoldingRef [0..1] (none = outside)
        +amount : Money (rounded, ADR-0009)
        +sourceId (the item that caused it)
        +tags : Set~String~ (ADR-0023)
    }
    Movement --> MovementKind
    class MovementKind {
        <<enumeration>>
        INCOME
        TAX
        DEDUCTION_CREDIT
        EXPENSE
        LOAN_ADVANCE
        LOAN_INTEREST
        LOAN_PAYMENT
        PREPAYMENT
        PURCHASE
        IMPROVEMENT
        SALE
        SELLING_COST
        PENALTY
        CONTRIBUTION
        EMPLOYER_MATCH
        WITHDRAWAL
        PENSION_PAYMENT
        RETURN
        APPRECIATION
        SWEEP
        TRANSFER
    }

    %% ---------- Tallies, goals, shortfalls ----------
    class TallySeries {
        +tallyId
        +values : Money[1..*] (one per period)
    }
    class GoalResult {
        +goalId
        +kind : REACH | MAINTAIN
        +status : MET | NOT_MET | HELD | VIOLATED
        +firstMet : Period [0..1]
        +firstViolated : Period [0..1]
    }
    class Shortfall {
        +period : Period
        +account : HoldingRef
        +amount : Money
        +cause : sourceId
    }
    class AppliedDefault {
        +path (e.g. items.house.sellingCostRate)
        +value
    }
