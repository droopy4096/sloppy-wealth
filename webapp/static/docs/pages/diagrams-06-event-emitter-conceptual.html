classDiagram
    direction LR

    namespace Input {
        class Scenario {
                <<input>>
                horizon, settings, jurisdiction
                tax rates, inflation, base date
                items: person, retirement, trigger, holdings, flows...
            }
        class ItemSpec {
                <<input: one form input>>
                type, id, fields
                host : ItemSpec [0..1] (modifier, D-E1)
            }
        class Validator {
                <<pure function>>
                validate(Scenario)
            }
        class Preparation {
                <<compiler.prepared>>
                upgrade (older forms), enabled.active (off items out),
                triggers.resolve (dates written into targets)
            }
    }

    namespace Build {
        class EmitterFactory {
                <<registry: @item_type, one per item type>>
                factory(item, ctx) Emitter[*]
                @scenario_setup hooks: retirement + tax, results
                compiler.compile_scenario builds Household + runner
            }
        class Emitter {
                <<abstract>>
                order : clock.Order number, class default, overridable, D-E2
                ordinal : holding slot or item index (tie-break)
                sub : int (host and modifiers at one position)
                source : item id
                schedule : Schedule
                due(month) bool
                emit(month, Household) Event[*]
            }
        class Schedule {
                <<value>>
                initial | one-shot | window | anniversary | every month
                may be anchored to retirement
            }
        class InitialEmitter {
                carry-in, account at start, pension creation
                negative order: once before month 1
            }
        class OneShotEmitter {
                purchase, opening, transfer, improvement,
                market adjustment
            }
        class RecurringEmitter {
                income, expense, contribution, withdrawal,
                loan servicing, growth, maintenance, pension
            }
        class ModifierEmitter {
                renewal, extra payment, lump sum, sale
                acts at its host's position (D-E1)
            }
        class ScenarioEmitter {
                retirement (one per person), tax step,
                results Recorder (CLOSE, emits no events)
            }
    }

    namespace Run {
        class ScenarioRunner {
                <<the loop>>
                sorts emitters by key = order, ordinal, sub, D-E2
                INITIAL step once, then for each month:
                household.begin_month (OPEN), then
                ask each due emitter in key order,
                apply its events at once
                registers Emitters only
            }
        class Event {
                <<abstract>>
                key : order, ordinal, sub
                source : ItemSpec
                apply(Household) Event[*] (follow-ups, D-E3)
            }
        class LifecycleEvent {
                creates or closes an Aggregate (D1)
                purchase, loan advance, opening, sale, payoff
            }
        class FlowEvent {
                moves money
                income, expense, payment, prepayment, contribution,
                withdrawal, transfer, maintenance, improvement, penalty
            }
        class ValuationEvent {
                changes a value, no cash
                growth, interest, revaluation, market adjustment
            }
        class StateEvent {
                changes household state
                retirement, renewal, tax step, flow adjustment,
                contribution room, trigger (log only)
            }
    }

    namespace State {
        class Household {
                <<main object>>
                month, covered mode, persons, jurisdiction
                move(kind, from, to, amount, source)
                begin_month(m)
            }
        class Person {
                id, name, retirement, retired
                tax rate, retirement tax rate
            }
        class Jurisdiction {
                <<plug-in package: CA, US (ADR-0071)>>
                account types and tax Treatment,
                ContributionLimits, own item types,
                on_contribute / on_return / on_withdraw
            }
        class Aggregate {
                <<abstract holding>>
                id, name, kind, balance
                lifecycle state
            }
        class Cash
        class Asset {
                RealAsset: RealEstate, Vehicle
            }
        class Liability {
                Mortgage (loan)
                secured by an Asset
            }
        class Account {
                InvestmentAccount: RRSP, TFSA, non-registered
            }
        class Entitlement {
                Pension (guaranteed)
            }
        class Rule {
                <<abstract, stateless>>
                tax, indexing, compounding, amortization,
                investment income, withdrawal tax,
                deduction, capital gains, pension value
            }
        class TaxAccumulator {
                per person: taxable, deductions,
                own-rate tax, carry-forward pool, monthly detail
            }
        class Ledger {
                Movement[*]: every money change
            }
    }

    namespace Output {
        class EventLog {
                every applied Event (D-E4)
                month, key, emitter, event, kind,
                source, cause, ledger range
            }
        class ResultAssembler {
                Recorder (CLOSE) + assemble
                rows, holdings detail, ledger, tax, events
            }
        class EngineAPI {
                <<state_machine/api.py, facade engine.py>>
                validate, run, run_detailed
            }
        class JsonAPI {
                <<webapp/api.py, stateless, ADR-0069>>
                commands (ops.py), run, views
            }
        class App {
                <<Flask pages, webapp/web>>
                form: one input per item,
                modifiers grouped under host;
                calls the JSON API only (web/client.py)
            }
    }

    %% ---------- Relationships ----------
    Scenario "1" *-- "*" ItemSpec
    ItemSpec --> ItemSpec : modifies host
    Validator ..> Scenario : checks
    Preparation ..> Scenario : prepares

    EmitterFactory ..> ItemSpec : reads
    EmitterFactory ..> Emitter : builds 1..n per item
    Emitter --> Schedule
    Emitter ..> Rule : decides with
    Emitter <|-- InitialEmitter
    Emitter <|-- OneShotEmitter
    Emitter <|-- RecurringEmitter
    Emitter <|-- ModifierEmitter
    Emitter <|-- ScenarioEmitter
    ModifierEmitter --> Emitter : host

    ScenarioRunner "1" o-- "*" Emitter : sorted by order
    ScenarioRunner --> Household : steps month by month
    ScenarioRunner ..> Event : applies
    ScenarioRunner --> EventLog : records
    Emitter ..> Event : emits

    Event <|-- LifecycleEvent
    Event <|-- FlowEvent
    Event <|-- ValuationEvent
    Event <|-- StateEvent
    Event ..> Household : mutates
    Event ..> Event : follow-ups

    Household "1" *-- "*" Aggregate : by holding slot
    Household "1" *-- "1" Ledger
    Household "1" *-- "1..*" Person
    Person "1" *-- "1" TaxAccumulator
    Household --> Jurisdiction : tax rules and limits
    Aggregate <|-- Cash
    Aggregate <|-- Asset
    Aggregate <|-- Liability
    Aggregate <|-- Account
    Aggregate <|-- Entitlement
    Aggregate --> Rule : references
    Liability --> Asset : secured by
    StateEvent ..> TaxAccumulator : tax step
    FlowEvent ..> Jurisdiction : account hooks

    ResultAssembler ..> Household : reads
    ResultAssembler ..> EventLog : reads
    EngineAPI ..> Validator
    EmitterFactory ..> Preparation : compile_scenario first
    EngineAPI ..> EmitterFactory
    EngineAPI ..> ScenarioRunner
    EngineAPI ..> ResultAssembler
    JsonAPI ..> EngineAPI : calls
    App ..> JsonAPI : calls
    App ..> Scenario : edits
